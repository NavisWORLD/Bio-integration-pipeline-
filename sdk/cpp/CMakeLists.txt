cmake_minimum_required(VERSION 3.16)
project(cosmos_synapse VERSION 0.2.0 LANGUAGES CXX)
add_library(cosmos_synapse STATIC src/c_api.cpp)
target_include_directories(cosmos_synapse PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(cosmos_synapse PUBLIC cxx_std_17)
add_executable(cosmos_synapse_test tests/test_synapse.cpp)
target_link_libraries(cosmos_synapse_test PRIVATE cosmos_synapse)
enable_testing()
add_test(NAME cosmos_synapse_golden COMMAND cosmos_synapse_test)
install(TARGETS cosmos_synapse ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
